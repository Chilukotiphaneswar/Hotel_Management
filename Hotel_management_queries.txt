CREATE DATABASE HotelManagement;
USE HotelManagement;
CREATE TABLE Customers (
    customer_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_name VARCHAR(100),
    phone VARCHAR(15),
3
    email VARCHAR(100)
);

CREATE TABLE Rooms (
    room_id INT PRIMARY KEY AUTO_INCREMENT,
    room_number INT UNIQUE,
    room_type VARCHAR(50),
    price DECIMAL(10,2)
);

CREATE TABLE Bookings (
    booking_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT,
    room_id INT,
    check_in DATE,
    check_out DATE,
    FOREIGN KEY (customer_id)
        REFERENCES Customers(customer_id),
    FOREIGN KEY (room_id)
        REFERENCES Rooms(room_id)
);

CREATE TABLE Payments (
    payment_id INT PRIMARY KEY AUTO_INCREMENT,
    booking_id INT,
    amount DECIMAL(10,2),
    payment_date DATE,
    payment_method VARCHAR(30),
    FOREIGN KEY (booking_id)
        REFERENCES Bookings(booking_id)
);
desc payments;
desc customers;
desc rooms;
desc bookings;

CREATE VIEW CustomerBookingDetails AS
SELECT
    c.customer_id,
    c.customer_name,
    c.phone,
    r.room_number,
    r.room_type,
    b.check_in,
    b.check_out
FROM Customers c
JOIN Bookings b
    ON c.customer_id = b.customer_id
JOIN Rooms r
    ON b.room_id = r.room_id;
SELECT * FROM CustomerBookingDetails;

CREATE VIEW AvailableRooms AS
SELECT
    r.room_id,
    r.room_number,
    r.room_type,
    r.price
FROM Rooms r
WHERE NOT EXISTS (
    SELECT 1
    FROM Bookings b
    WHERE b.room_id = r.room_id
      AND CURDATE() >= b.check_in
      AND CURDATE() < b.check_out
);
SELECT * FROM AvailableRooms;

CREATE VIEW BookingPaymentSummary AS
SELECT
    b.booking_id,
    c.customer_name,
    r.room_number,
    r.price AS room_price_per_day,
    b.check_in,
    b.check_out,
    COALESCE(SUM(p.amount), 0) AS total_paid
FROM Bookings b
JOIN Customers c
    ON b.customer_id = c.customer_id
JOIN Rooms r
    ON b.room_id = r.room_id
LEFT JOIN Payments p
    ON b.booking_id = p.booking_id
GROUP BY
    b.booking_id,
    c.customer_name,
    r.room_number,
    r.price,
    b.check_in,
    b.check_out;

    DELIMITER //
CREATE PROCEDURE GetCustomerBookings(IN p_customer_id INT)
BEGIN
    SELECT
        c.customer_name,
        c.phone,
        r.room_number,
        r.room_type,
        b.check_in,
        b.check_out
    FROM Customers c
    JOIN Bookings b
        ON c.customer_id = b.customer_id
    JOIN Rooms r
        ON b.room_id = r.room_id
    WHERE c.customer_id = p_customer_id;
END //
DELIMITER ;
CALL GetCustomerBookings(1);


DELIMITER //
CREATE PROCEDURE CheckAvailableRooms(IN p_date DATE)
BEGIN
    SELECT
        room_id,
        room_number,
        room_type,
        price
    FROM Rooms
    WHERE room_id NOT IN (
        SELECT room_id
        FROM Bookings
        WHERE p_date >= check_in
          AND p_date < check_out
    );
END //
DELIMITER ;
CALL CheckAvailableRooms('2026-11-01');



DELIMITER //
CREATE PROCEDURE GetBookingPaymentDetails(IN p_booking_id INT)
BEGIN
    SELECT
        b.booking_id,
        c.customer_name,
        r.room_number,
        COALESCE(SUM(p.amount), 0) AS total_paid
    FROM Bookings b
    JOIN Customers c
        ON b.customer_id = c.customer_id
    JOIN Rooms r
        ON b.room_id = r.room_id
    LEFT JOIN Payments p
        ON b.booking_id = p.booking_id
    WHERE b.booking_id = p_booking_id
    GROUP BY
        b.booking_id,
        c.customer_name,
        r.room_number;
END //
DELIMITER ;
CALL GetBookingPaymentDetails(1);

DELIMITER //
CREATE TRIGGER CheckRoomPrice
BEFORE INSERT ON Rooms
FOR EACH ROW
BEGIN
    IF NEW.price <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Room price must be greater than zero';
    END IF;
END //
DELIMITER ;
INSERT INTO Rooms (room_number, room_type, price)
VALUES (109, 'Deluxe', 2500);

DELIMITER //
CREATE TRIGGER CheckBookingDates
BEFORE INSERT ON Bookings
FOR EACH ROW
BEGIN
    IF NEW.check_out <= NEW.check_in THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Check-out date must be after check-in date';
    END IF;
END //
DELIMITER ;

(customer_id, room_id, check_in, check_out)
